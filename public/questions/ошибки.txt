  <script>
    let playerCount = 0;

    // Следим за мышкой
    document.addEventListener('mousemove', (e) => {
      const eyes = document.querySelectorAll('.eye');
      const rect = document.querySelector('.printer').getBoundingClientRect();
      const centerX = rect.left + rect.width / 2;
      const centerY = rect.top + rect.height / 2;

      eyes.forEach(eye => {
        const eyeRect = eye.getBoundingClientRect();
        const eyeX = eyeRect.left + eyeRect.width / 2;
        const eyeY = eyeRect.top + eyeRect.height / 2;

        // Угол между глазом и курсором
        const angle = Math.atan2(e.clientY - eyeY, e.clientX - eyeX);
        // Смещение зрачка (не слишком далеко)
        const dx = Math.cos(angle) * 2;
        const dy = Math.sin(angle) * 2;

        const pupil = eye.querySelector('.pupil');
        pupil.style.transform = `translate(${dx}px, ${dy}px)`;
      });
    });

    document.getElementById('joinBtn').addEventListener('click', () => {
      playerCount++;

      const paper = document.createElement('div');
      paper.className = 'paper';
      paper.textContent = playerCount;

      document.querySelector('.printer').appendChild(paper);

      setTimeout(() => {
        paper.classList.add('printed');
      }, 50);
    });
  </script>
