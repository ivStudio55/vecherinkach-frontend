  <script>
    (function () {
      "use strict";

      const COUNTRIES = [
        "Россия", "Япония", "Канада", "Бразилия", "Германия", "Испания", "Италия", "Австралия",
        "Индия", "Мексика", "Норвегия", "Аргентина", "Франция", "Польша", "Турция", "Египет",
        "Швеция", "Финляндия", "Чили", "Португалия", "Нидерланды", "Швейцария", "Австрия", "Греция",
        "Таиланд", "Вьетнам", "Индонезия", "Марокко", "ЮАР", "Новая Зеландия", "Ирландия", "Исландия",
        "США", "Китай", "Корея", "Сингапур", "ОАЭ", "Саудовская Аравия", "Украина", "Казахстан"
      ];

      const gridElement = document.getElementById("grid");
      const actionButton = document.getElementById("actionButton");
      const liveRegion = document.getElementById("liveRegion");
      const roundIndexElement = document.getElementById("roundIndex");

      let phase = "entering";
      let roundIndex = 1;
      let currentOptions = [];
      let correctIndex = 0;

      function prefersReducedMotion() {
        return window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
      }

      function setLive(text) {
        liveRegion.textContent = text;
      }

      function setButtonBusy(isBusy) {
        actionButton.setAttribute("aria-busy", String(isBusy));
        actionButton.disabled = !!isBusy;
      }

      function randomBetween(min, max) {
        return min + Math.random() * (max - min);
      }

      function sampleUnique(array, count) {
        const copy = array.slice();
        for (let i = copy.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [copy[i], copy[j]] = [copy[j], copy[i]];
        }
        return copy.slice(0, count);
      }

      function clearGrid() {
        while (gridElement.firstChild) gridElement.removeChild(gridElement.firstChild);
      }

      function createTile(text, positionIndex) {
        const tile = document.createElement("div");
        tile.className = "tile pos-" + positionIndex;
        tile.setAttribute("role", "button");
        tile.setAttribute("tabindex", "0");
        tile.setAttribute("aria-disabled", "true");
        tile.setAttribute("aria-label", "Вариант: " + text + ". Выбор отключён в демо.");

        const label = document.createElement("div");
        label.textContent = text;

        const hint = document.createElement("div");
        hint.className = "hint";
        hint.textContent = "вариант";

        tile.appendChild(label);
        tile.appendChild(hint);
        tile.style.setProperty("--enter-delay", (positionIndex * 100) + "ms");
        return tile;
      }

      function renderRound() {
        clearGrid();

        currentOptions = sampleUnique(COUNTRIES, 4);
        correctIndex = Math.floor(Math.random() * 4);

        const tiles = [];
        for (let i = 0; i < 4; i++) {
          const tile = createTile(currentOptions[i], i);
          tile.dataset.index = String(i);
          tile.dataset.isCorrect = String(i === correctIndex);
          tiles.push(tile);
          gridElement.appendChild(tile);
        }

        phase = "entering";
        setButtonBusy(true);

        requestAnimationFrame(() => {
          requestAnimationFrame(() => {
            tiles.forEach(t => t.classList.add("enter"));

            if (prefersReducedMotion()) {
              tiles.forEach(t => t.classList.remove("enter"));
              phase = "ready";
              setButtonBusy(false);
              actionButton.textContent = "Запустить анимацию";
              setLive("Новые варианты загружены.");
              return;
            }

            const total = 850 + 100 * 3 + 60;
            setTimeout(() => {
              tiles.forEach(t => t.classList.remove("enter"));
              phase = "ready";
              setButtonBusy(false);
              actionButton.textContent = "Запустить анимацию";
              setLive("Варианты готовы. Нажмите кнопку.");
            }, total);
          });
        });
      }

      function applyRandomFallVars(tile) {
        const dx = Math.round(randomBetween(-180, 180));
        const rot = Math.round(randomBetween(-200, 200));
        const dur = Math.round(randomBetween(800, 1300));
        const delay = Math.round(randomBetween(0, 220));
        const drop = Math.round(randomBetween(500, 680));

        tile.style.setProperty("--dx", dx + "px");
        tile.style.setProperty("--rot", rot + "deg");
        tile.style.setProperty("--dur", dur + "ms");
        tile.style.setProperty("--delay", delay + "ms");
        tile.style.setProperty("--drop", drop + "px");

        return { dur, delay };
      }

      function fallIncorrectTiles() {
        if (phase !== "ready") return;

        const tiles = Array.from(gridElement.querySelectorAll(".tile"));
        if (tiles.length !== 4) return;

        phase = "falling";
        setButtonBusy(true);

        let maxEnd = 0;
        let fallenCount = 0;

        tiles.forEach(tile => {
          const isCorrect = tile.dataset.isCorrect === "true";
          if (isCorrect) return;

          const timings = applyRandomFallVars(tile);
          const end = timings.delay + timings.dur;
          maxEnd = Math.max(maxEnd, end);

          tile.classList.add("fall");
          fallenCount++;
        });

        if (prefersReducedMotion()) {
          tiles.forEach(tile => {
            const isCorrect = tile.dataset.isCorrect === "true";
            if (!isCorrect) {
              tile.classList.add("invisible");
            }
          });
          phase = "afterFall";
          setButtonBusy(false);
          actionButton.textContent = "Дальше";
          setLive("Неправильные варианты скрыты. Нажмите Дальше.");
          return;
        }

        // Важно: НЕ удаляем элементы из DOM!
        // После анимации падения просто делаем их невидимыми
        setTimeout(() => {
          tiles.forEach(tile => {
            const isCorrect = tile.dataset.isCorrect === "true";
            if (!isCorrect) {
              tile.classList.remove("fall");
              tile.classList.add("invisible");
            }
          });

          phase = "afterFall";
          setButtonBusy(false);
          actionButton.textContent = "Дальше";
          setLive("Неправильные варианты упали. Остался один. Нажмите Дальше.");
        }, maxEnd + 40);

        if (fallenCount === 0) {
          phase = "afterFall";
          setButtonBusy(false);
          actionButton.textContent = "Дальше";
        }
      }

      function hideCorrectThenNextRound() {
        if (phase !== "afterFall") return;

        const correctTile = gridElement.querySelector('.tile[data-is-correct="true"]');
        if (!correctTile) {
          nextRound();
          return;
        }

        phase = "hiding";
        setButtonBusy(true);
        correctTile.classList.add("hide");

        if (prefersReducedMotion()) {
          nextRound();
          return;
        }

        setTimeout(() => {
          nextRound();
        }, 600 + 40);
      }

      function nextRound() {
        roundIndex += 1;
        roundIndexElement.textContent = String(roundIndex);
        renderRound();
      }

      function onActionClick() {
        if (phase === "ready") {
          fallIncorrectTiles();
          return;
        }
        if (phase === "afterFall") {
          hideCorrectThenNextRound();
          return;
        }
      }

      actionButton.addEventListener("click", onActionClick);

      gridElement.addEventListener("keydown", (event) => {
        const target = event.target;
        if (!target.classList.contains("tile")) return;
        if (event.key === "Enter" || event.key === " ") {
          event.preventDefault();
          setLive("Выбор вариантов отключён. Используйте кнопку.");
        }
      });

      roundIndexElement.textContent = String(roundIndex);
      setLive("Загрузка вариантов...");
      renderRound();
    })();
  </script>
